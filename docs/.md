# Phase 14.2 Implementation Guide: Frontend Premium Components

## Overview

This guide provides step-by-step instructions for implementing Phase 14.2, which builds the premium component library and updates the renderer to use them.

**Deliverables:**
- 10 premium React components
- Component registry
- Updated renderer with componentId support
- Responsive design and animations
- Accessibility compliance

**Duration:** 4-5 days

## Step 1: Create Component Registry

### 1.1 Create `apps/web/src/lib/component-registry.ts`

```typescript
import { ComponentType } from 'react';
import type { SiteSection } from '@/lib/types';

// Import all premium components (we'll create these next)
import HeroSplitEditorial from '@/components/premium/HeroSplitEditorial';
import HeroMediaLed from '@/components/premium/HeroMediaLed';
import ServicesDynamicBento from '@/components/premium/ServicesDynamicBento';
import ServicesGrid from '@/components/premium/ServicesGrid';
import ProofCarousel from '@/components/premium/ProofCarousel';
import ProofGrid from '@/components/premium/ProofGrid';
import GalleryMasonry from '@/components/premium/GalleryMasonry';
import ProcessTimeline from '@/components/premium/ProcessTimeline';
import AboutStatement from '@/components/premium/AboutStatement';
import CTAPanel from '@/components/premium/CTAPanel';

export interface PremiumComponentProps {
  section: SiteSection;
  brandTokens: any;
  paletteMode: string;
  dna: any;
}

export type PremiumComponentType = ComponentType<PremiumComponentProps>;

export const COMPONENT_REGISTRY: Record<string, PremiumComponentType> = {
  'hero-split-editorial': HeroSplitEditorial,
  'hero-media-led': HeroMediaLed,
  'services-dynamic-bento': ServicesDynamicBento,
  'services-grid': ServicesGrid,
  'proof-carousel': ProofCarousel,
  'proof-grid': ProofGrid,
  'gallery-masonry': GalleryMasonry,
  'process-timeline': ProcessTimeline,
  'about-statement': AboutStatement,
  'cta-panel': CTAPanel,
};

export const COMPONENT_IDS = Object.keys(COMPONENT_REGISTRY);

export function getComponent(componentId: string): PremiumComponentType | null {
  return COMPONENT_REGISTRY[componentId] || null;
}

export function isValidComponentId(componentId: string): boolean {
  return componentId in COMPONENT_REGISTRY;
}
```

import { motion } from 'framer-motion';
import type { SiteSection } from '@/lib/types';

interface Props {
  section: SiteSection;
  brandTokens: any;
  paletteMode: string;
  dna: any;
}

export default function HeroSplitEditorial({
  section,
  brandTokens,
  paletteMode,
  dna,
}: Props) {
  const textColor = paletteMode === 'light' ? 'text-slate-900' : 'text-slate-50';
  const mutedColor = paletteMode === 'light' ? 'text-slate-600' : 'text-slate-400';

  return (
    <section className="relative py-20 md:py-32 overflow-hidden">
      <div className="mx-auto max-w-7xl px-4 md:px-8">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-12 md:gap-16 items-center">
          {/* Left: Content */}
          <motion.div
            initial={{ opacity: 0, x: -20 }}
            whileInView={{ opacity: 1, x: 0 }}
            transition={{ duration: 0.6 }}
            viewport={{ once: true }}
            className="flex flex-col gap-6"
          >
            <div>
              <h2
                className={`text-5xl md:text-6xl font-bold leading-tight tracking-tight ${textColor}`}
                style={{ fontFamily: dna.fontFamily }}
              >
                {section.headline}
              </h2>
            </div>

            {section.body && (
              <p className={`text-lg md:text-xl leading-relaxed ${mutedColor}`}>
                {section.body}
              </p>
            )}

            {section.items && section.items.length > 0 && (
              <ul className="space-y-3 pt-4">
                {section.items.slice(0, 4).map((item, idx) => (
                  <motion.li
                    key={idx}
                    initial={{ opacity: 0, x: -10 }}

            initial={{ opacity: 0, x: 20 }}

            viewport={{ once: true }}

### 2.6 Create `ProofCarousel.tsx`

```typescript
'use client';

import { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import type { SiteSection } from '@/lib/types';

interface Props {
  section: SiteSection;
  brandTokens: any;
  paletteMode: string;
  dna: any;
}

export default function ProofCarousel({
  section,
  brandTokens,
  paletteMode,
  dna,
}: Props) {
  const [activeIndex, setActiveIndex] = useState(0);
  const textColor = paletteMode === 'light' ? 'text-slate-900' : 'text-slate-50';
  const mutedColor = paletteMode === 'light' ? 'text-slate-600' : 'text-slate-400';
  const panelBg = paletteMode === 'light'
    ? 'bg-white/70 border-slate-200'
    : 'bg-white/[0.045] border-white/10';

  const items = section.items || [];
  const displayItems = items.slice(0, 5);

  if (displayItems.length === 0) return null;

  return (
    <section className="relative py-20 md:py-32">
      <div className="mx-auto max-w-7xl px-4 md:px-8">
        {/* Header */}
        <motion.div
          initial={{ opacity: 0, y: -20 }}
          whileInView={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.5 }}
          viewport={{ once: true }}
          className="mb-16 max-w-2xl"
        >
          <h2
            className={`text-4xl md:text-5xl font-bold leading-tight tracking-tight ${textColor}`}
            style={{ fontFamily: dna.fontFamily }}
          >
            {section.headline}
          </h2>
        </motion.div>

        {/* Carousel */}
        <div className="relative">
          <div className={`p-12 rounded-2xl border ${panelBg} min-h-64`}>
            <AnimatePresence mode="wait">
              <motion.div
                key={activeIndex}
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -10 }}
                transition={{ duration: 0.3 }}
                className="text-center"
              >
                <p className={`text-lg md:text-xl leading-relaxed ${textColor}`}>
                  {displayItems[activeIndex]}
                </p>
              </motion.div>
            </AnimatePresence>
          </div>

          {/* Navigation */}
          <div className="flex justify-center gap-2 mt-8">
            {displayItems.map((_, idx) => (
              <button
                key={idx}
                onClick={() => setActiveIndex(idx)}
                className={`w-2 h-2 rounded-full transition-all ${
                  idx === activeIndex
                    ? 'w-8'
                    : 'opacity-50 hover:opacity-75'
                }`}
                style={{
                  backgroundColor: idx === activeIndex ? dna.accentHue : '#999',
                }}
                aria-label={`Go to item ${idx + 1}`}
              />
            ))}
          </div>
        </div>
      </div>
    </section>
  );
}
```

'use client';

import { motion } from 'framer-motion';
import type { SiteSection } from '@/lib/types';

interface Props {
  section: SiteSection;
  brandTokens: any;
  paletteMode: string;
  dna: any;
}

export default function CTAPanel({
  section,
  brandTokens,
  paletteMode,
  dna,
}: Props) {
  const textColor = paletteMode === 'light' ? 'text-slate-900' : 'text-slate-50';
  const mutedColor = paletteMode === 'light' ? 'text-slate-600' : 'text-slate-400';
  const panelBg = paletteMode === 'light'
    ? 'bg-white/70 border-slate-200'
    : 'bg-white/[0.045] border-white/10';

  return (
    <section className="relative py-20 md:py-32 overflow-hidden">
      <div className="mx-auto max-w-5xl px-4 md:px-8">
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          whileInView={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.6 }}
          viewport={{ once: true }}
          className={`relative p-12 md:p-16 rounded-2xl border ${panelBg} overflow-hidden`}
        >
          {/* Background accent */}
          <div
            className="absolute -right-24 -top-24 w-96 h-96 rounded-full opacity-10"
            style={{ backgroundColor: dna.accentHue }}
          />

          <div className="relative z-10 text-center max-w-3xl mx-auto">
            <h2
              className={`text-4xl md:text-5xl font-bold leading-tight tracking-tight ${textColor}`}
              style={{ fontFamily: dna.fontFamily }}
            >
              {section.headline}
            </h2>

            {section.body && (
              <p className={`mt-6 text-lg md:text-xl leading-relaxed ${mutedColor}`}>
                {section.body}
              </p>
            )}

            {section.items && section.items.length > 0 && (
              <div className="mt-10 flex flex-wrap justify-center gap-4">
                {section.items.slice(0, 3).map((item, idx) => (
                  <motion.button
                    key={idx}
                    initial={{ opacity: 0, y: 10 }}
                    whileInView={{ opacity: 1, y: 0 }}
                    transition={{ duration: 0.4, delay: idx * 0.1 }}
                    viewport={{ once: true }}
                    className={`px-6 py-3 rounded-full font-semibold transition-all ${
                      idx === 0
                        ? `text-white`
                        : `border ${
                            paletteMode === 'light'
                              ? 'border-slate-300 text-slate-900'
                              : 'border-white/20 text-white'
                          }`
                    }`}
                    style={{
                      backgroundColor: idx === 0 ? dna.accentHue : 'transparent',
                    }}
                  >
                    {item}
                  </motion.button>
                ))}
              </div>
            )}
          </div>
        </motion.div>
      </div>
    </section>
  );
}
